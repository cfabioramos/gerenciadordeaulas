-- ============================
-- TABELA: Tipo_Aula
-- ============================
CREATE TABLE tipo_aula (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL
);

-- ============================
-- TABELA: Ciclo
-- ============================
CREATE TABLE ciclo (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    data_inicio DATE NOT NULL,
    data_fim DATE NOT NULL
);

-- ============================
-- TABELA: Programa_Aula
-- Cada programa de aula pertence a um ciclo
-- ============================
CREATE TABLE programa_aula (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    ciclo_id INT NOT NULL,
    tipoaula_id INT,
    CONSTRAINT fk_programaaula_ciclo FOREIGN KEY (ciclo_id) REFERENCES ciclo (id) ON DELETE RESTRICT,
    CONSTRAINT fk_programaaula_tipoaula FOREIGN KEY (tipoaula_id) REFERENCES tipo_aula (id) ON DELETE RESTRICT
);

-- ============================
-- TABELA: Aula
-- Cada aula pertence a um programa de aula
-- ============================
CREATE TABLE aula (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    data TIMESTAMP NOT NULL,
    programaaula_id INT NOT NULL,
    CONSTRAINT fk_aula_programaaula FOREIGN KEY (programaaula_id) REFERENCES programa_aula (id) ON DELETE RESTRICT
);

-- ============================
-- TABELA: Aluno
-- ============================
CREATE TABLE aluno (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL
);

-- ============================
-- TABELA: Matrícula
-- Relaciona aluno ↔ ciclo (* x *)
-- ============================
CREATE TABLE matricula (
    id SERIAL PRIMARY KEY,
    aluno_id INT NOT NULL,
    ciclo_id INT NOT NULL,
    data DATE NOT NULL DEFAULT CURRENT_DATE,
    fl_ativo BOOLEAN,
    valor DECIMAL(10, 2),
    valor_mensalidade DECIMAL(10, 2),
    dia_vencimento INT,
    CONSTRAINT fk_matricula_aluno FOREIGN KEY (aluno_id) REFERENCES aluno (id) ON DELETE CASCADE,
    CONSTRAINT fk_matricula_ciclo FOREIGN KEY (ciclo_id) REFERENCES ciclo (id) ON DELETE CASCADE,
    CONSTRAINT uq_matricula UNIQUE (aluno_id, ciclo_id) -- evita duplicidade
);

-- ============================
-- TABELA: Presença
-- Relaciona matrícula ↔ aula (* x *)
-- ============================
CREATE TABLE presenca (
    id SERIAL PRIMARY KEY,
    aula_id INT NOT NULL,
    matricula_id INT NOT NULL,
    CONSTRAINT fk_presenca_aula FOREIGN KEY (aula_id) REFERENCES aula (id) ON DELETE RESTRICT,
    CONSTRAINT fk_presenca_matricula FOREIGN KEY (matricula_id) REFERENCES matricula (id),
    CONSTRAINT uq_presenca UNIQUE (aula_id, matricula_id) -- evita duplicidade
);

-- ============================
-- TABELA: Pagamento
-- ============================
CREATE TABLE pagamento (
    id BIGSERIAL PRIMARY KEY,
    valor DECIMAL(10, 2) NOT NULL,
    data DATE NOT NULL
);

-- ============================
-- TABELA: Pagamento_Matrícula
-- ============================
CREATE TABLE pagamento_matricula (
    id BIGSERIAL PRIMARY KEY,
    valor_mensalidade_dia DECIMAL(10, 2) NOT NULL,
    matricula_id BIGINT NOT NULL,
    pagamento_id BIGINT NOT NULL,
    CONSTRAINT fk_pagamento_matricula_matricula FOREIGN KEY (matricula_id) REFERENCES matricula (id) ON DELETE CASCADE,
    CONSTRAINT fk_pagamento_matricula_pagamento FOREIGN KEY (pagamento_id) REFERENCES pagamento (id) ON DELETE CASCADE
);

-- ============================
-- TABELA: Usuario
-- ============================
CREATE TABLE usuario (
    id SERIAL PRIMARY KEY,
    username VARCHAR(100) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    admin BOOLEAN NOT NULL,
    theme VARCHAR(50) DEFAULT 'nordeste' NOT NULL,
    nome VARCHAR(100)
);
